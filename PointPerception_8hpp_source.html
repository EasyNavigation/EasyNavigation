54struct hash<std::tuple<int, int, int>>
55{
59 std::size_t operator()(const std::tuple<int, int, int> & key) const
60 {
61 std::size_t h1 = std::hash<int>()(std::get<0>(key));
62 std::size_t h2 = std::hash<int>()(std::get<1>(key));
63 std::size_t h3 = std::hash<int>()(std::get<2>(key));
64 return h1 ^ (h2 << 1) ^ (h3 << 2);
65 }
66};

183 void integrate_pending_perceptions()
184 {
185 std::lock_guard<std::mutex> lock(mutex_);
186 // Access TF buffer singleton (already initialized somewhere with a clock)
187 auto tf_buffer_ptr = RTTFBuffer::getInstance();
188 auto & tf_buffer = *tf_buffer_ptr;
189 const auto tf_info = tf_buffer.get_tf_info();
190 const std::string & robot_frame = tf_info.robot_frame;
191
192 // ------------------------------------------------------------------
193 // 1. Push pending perception into the circular buffer exactly once.
194 // ------------------------------------------------------------------
195 if (pending_available_) {
196 PointPerceptionBufferType pending_item;
197 pending_item.data = std::move(pending_cloud_); // avoid deep copy
198 pending_item.frame = pending_frame_;
199 pending_item.stamp = pending_stamp_;
200
201 buffer.push(std::move(pending_item));
202 pending_available_ = false;
203 }
204
205 const std::size_t count = buffer.size();
206 if (count == 0) {
207 // No candidates at all: keep current visible state as is.
208 return;
209 }
210
211 // ------------------------------------------------------------------
212 // 2. Drain the circular buffer into a temporary vector so we can
213 // inspect all items and then rebuild the buffer.
214 // ------------------------------------------------------------------
215 std::vector<PointPerceptionBufferType> items;
216 items.reserve(count);
217
218 for (std::size_t i = 0; i < count; ++i) {
219 PointPerceptionBufferType item;
220 if (!buffer.pop(item)) {
221 break; // Defensive guard if pop() fails unexpectedly.
222 }
223 items.push_back(std::move(item));
224 }
225
226 if (items.empty()) {
227 // Nothing recovered from buffer: keep visible state untouched.
228 return;
229 }
230
231 // ------------------------------------------------------------------
232 // 3. Find indices:
233 // - newest_idx: newest perception by timestamp (regardless of TF),
234 // - newest_valid_idx: newest perception that has a valid TF.
235 //
236 // IMPORTANT: store only indices to avoid copying point clouds
237 // during the scan.
238 // ------------------------------------------------------------------
239 std::optional<std::size_t> newest_idx;
240 std::optional<std::size_t> newest_valid_idx;
241
242 for (std::size_t i = 0; i < items.size(); ++i) {
243 const auto & item = items[i];
244
245 // Track newest item overall (used when no TF is valid).
246 if (!newest_idx || item.stamp > items[*newest_idx].stamp) {
247 newest_idx = i;
248 }
249
250 bool has_tf = false;
251 try {
252 has_tf = tf_buffer.canTransform(
253 robot_frame,
254 item.frame,
255 tf2_ros::fromMsg(item.stamp));
256 } catch (...) {
257 // Any TF exception is treated as "no valid TF" for this item.
258 has_tf = false;
259 }
260
261 if (has_tf) {
262 if (!newest_valid_idx || item.stamp > items[*newest_valid_idx].stamp) {
263 newest_valid_idx = i;
264 }
265 }
266 }
267
268 // ------------------------------------------------------------------
269 // 4. Update visible state BEFORE moving items back into the buffer.
270 // This guarantees that `data` corresponds to:
271 // - the newest TF-valid item if any exists, otherwise
272 // - the newest item overall.
273 // ------------------------------------------------------------------
274 if (newest_valid_idx) {
275 const auto & sel = items[*newest_valid_idx];
276 data = sel.data; // single deep copy (intentional)
277 frame_id = sel.frame;
278 stamp = sel.stamp;
279 valid = true; // "valid" means "usable / not too old", not "TF ok"
280 new_data = true;
281 } else if (newest_idx) {
282 const auto & sel = items[*newest_idx];
283 data = sel.data; // single deep copy (intentional)
284 frame_id = sel.frame;
285 stamp = sel.stamp;
286 valid = true;
287 new_data = true;
288 } else {
289 // Defensive: should not happen because items is non-empty.
290 return;
291 }
292
293 // ------------------------------------------------------------------
294 // 5. Rebuild the circular buffer from scratch (move-only, no copies).
295 // ------------------------------------------------------------------
296 buffer.clear();
297
298 if (newest_valid_idx) {
299 // Keep the newest TF-valid item and any newer items (even if TF is not yet available).
300 const rclcpp::Time cutoff_stamp = items[*newest_valid_idx].stamp;
301
302 for (auto & item : items) {
303 if (item.stamp >= cutoff_stamp) {
304 buffer.push(std::move(item));
305 }
306 }
307 } else {
308 // No TF-valid items: keep everything in case TF arrives later.
309 for (auto & item : items) {
310 buffer.push(std::move(item));
311 }
312 }
313 }
314
315protected:
316 mutable std::mutex mutex_;
317 CircularBuffer<PointPerceptionBufferType> buffer{10};
318};
