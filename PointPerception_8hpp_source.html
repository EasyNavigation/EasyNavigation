143PointPerceptions get_point_perceptions(std::vector<PerceptionPtr> & perceptionptr)
144{
145 PointPerceptions ret;
146
147 for (auto & ptr : perceptionptr) {
148 if (!ptr.perception) {
149 continue;
150 }
151
152 auto point_ptr = std::dynamic_pointer_cast<PointPerception>(ptr.perception);
153 if (!point_ptr) {
154 continue;
155 }
156
157 ret.push_back(point_ptr);
158 }
159
160 return ret;
161}
