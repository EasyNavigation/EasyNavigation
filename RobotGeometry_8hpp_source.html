86template<typename NodeT>
87RobotGeometry get_robot_geometry(NodeT & node, const LegacyRobotGeometryNames & legacy = {})
88{
89 auto registry = RobotGeometryRegistry::getInstance();
90 RobotGeometry geometry = registry->get_geometry();
91
92 auto apply = [&](const std::string & name, const std::string & field, double & value) {
93 if (name.empty()) {
94 return;
95 }
96 const auto & overrides = node.get_node_parameters_interface()->get_parameter_overrides();
97 if (overrides.count(name) == 0 && !node.has_parameter(name)) {
98 return;
99 }
100 double legacy_value = value;
101 declare_parameter_if_absent(node, name, legacy_value);
102 node.get_parameter(name, legacy_value);
103
104 const auto replacement = "system_node.robot_geometry." + field;
105 if (registry->is_configured(field)) {
106 RCLCPP_WARN(
107 node.get_logger(), "'%s' is deprecated and ignored: '%s' takes precedence",
108 name.c_str(), replacement.c_str());
109 return;
110 }
111 RCLCPP_WARN(
112 node.get_logger(), "'%s' is deprecated: configure '%s' instead. It will stop working soon.",
113 name.c_str(), replacement.c_str());
114 value = legacy_value;
115 };
116
117 apply(legacy.radius, "radius", geometry.radius);
118 apply(legacy.inscribed_radius, "inscribed_radius", geometry.inscribed_radius);
119 apply(legacy.height, "height", geometry.height);
120 return geometry;
121}
