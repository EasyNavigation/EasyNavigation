84template<typename T = PerceptionBase>
85inline std::vector<std::shared_ptr<T>>
86get_perceptions(const std::vector<PerceptionBasePtr> & src)
87{
88 static_assert(
89 std::is_base_of_v<PerceptionBase, T>,
90 "T must inherit from PerceptionBase");
91
92 std::vector<std::shared_ptr<T>> out;
93 out.reserve(src.size());
94
95 for (const auto & perception : src) {
96 if (!perception) {continue;}
97 // Homogeneous by derived type: include only successful casts
98 if (auto p = std::dynamic_pointer_cast<T>(perception)) {
99 out.push_back(std::move(p));
100 }
101 }
102 return out;
103}
