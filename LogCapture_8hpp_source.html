58 std::vector<std::string> find(
59 const std::vector<std::string> & parts, int severity = RCUTILS_LOG_SEVERITY_WARN) const
60 {
61 std::lock_guard<std::mutex> lock(mutex());
62 std::vector<std::string> found;
63 for (const auto & message : messages()) {
64 if (severity != -1 && message.severity != severity) {
65 continue;
66 }
67 bool all = true;
68 for (const auto & part : parts) {
69 all = all && message.text.find(part) != std::string::npos;
70 }
71 if (all) {
72 found.push_back(message.text);
73 }
74 }
75 return found;
76 }
77
79 size_t count(
80 const std::vector<std::string> & parts, int severity = RCUTILS_LOG_SEVERITY_WARN) const
81 {
82 return find(parts, severity).size();
83 }
84
85private:
86 static void handler(
87 const rcutils_log_location_t *, int severity, const char *, rcutils_time_point_value_t,
88 const char * format, va_list * args)
89 {
90 va_list copy;
91 va_copy(copy, *args);
92 char buffer[2048];
93 vsnprintf(buffer, sizeof(buffer), format, copy);
94 va_end(copy);
95 std::lock_guard<std::mutex> lock(mutex());
96 messages().push_back({severity, buffer});
97 }
98
99 static std::mutex & mutex()

101 static std::mutex m;

103 }
104
105 static std::vector<Message> & messages()
106 {
107 static std::vector<Message> m;
108 return m;
109 }

111 rcutils_logging_output_handler_t previous_;
112};
