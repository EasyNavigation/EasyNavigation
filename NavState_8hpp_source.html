63inline std::string stacktrace(std::size_t skip = 1, std::size_t max_frames = 64)
64{
65 void * buf[128];
66 const int n = backtrace(buf, static_cast<int>(std::min<std::size_t>(max_frames, 128)));
67 char ** syms = backtrace_symbols(buf, n);
68 std::ostringstream oss;
69 for (int i = static_cast<int>(skip); i < n; ++i) {
70 oss << "#" << (i - static_cast<int>(skip)) << " " << syms[i] << "\n";
71 }
72 std::free(syms);
73 return oss.str();
74}

107 template<typename T>
108 void set(const std::string & key, const T & value)
109 {
110 std::lock_guard<std::mutex> lock(mutex_);
111 auto it = values_.find(key);
112
113 if (it == values_.end()) {
114 values_[key] = std::make_shared<T>(value);
115 types_[key] = typeid(T).hash_code();
116 } else {
117 if (types_[key] != typeid(T).hash_code()) {
118 std::ostringstream oss;
119 oss << "Type mismatch in set(\"" << key << "\")\n"
120 << " expected(hash): " << types_[key] << "\n"
121 << " provided : " << demangle(typeid(T).name()) << "\n"
122 << "Backtrace:\n" << stacktrace(1);
123 throw std::runtime_error(oss.str());
124 }
125
126 auto ptr = std::static_pointer_cast<T>(it->second);
127 *ptr = value;
128 }
129 }

172 template<typename T>
173 const T & get(const std::string & key) const
174 {
175 std::lock_guard<std::mutex> lock(mutex_);
176 auto it = values_.find(key);
177
178 if (it == values_.end()) {
179 throw std::runtime_error("Key not found in get: " + key);
180 }
181
182 if (types_.at(key) != typeid(T).hash_code()) {
183 throw std::runtime_error("Type mismatch in get for key: " + key);
184 }
185
186 auto ptr = std::static_pointer_cast<T>(it->second);
187 return *ptr;
188 }

202 using AnyPrinter = std::function<std::string(std::shared_ptr<void>)>;

208 template<typename T>
209 static void register_printer(std::function<std::string(const T &)> printer)
210 {
211 auto wrapper = [printer](std::shared_ptr<void> base_ptr) -> std::string {
212 auto typed_ptr = std::static_pointer_cast<T>(base_ptr);
213 return printer(*typed_ptr);
214 };
215 type_printers_[typeid(T).hash_code()] = wrapper;
216 }

223 std::string debug_string() const
224 {
225 std::stringstream ss;
226 for (const auto & kv : values_) {
227 ss << kv.first << " = ";
228 auto ptr = kv.second;
229 if (ptr) {
230 auto type_it = types_.find(kv.first);
231 if (type_it != types_.end()) {
232 auto printer_it = type_printers_.find(type_it->second);
233 if (printer_it != type_printers_.end()) {
234 ss << "[" << ptr.get() << "] : " << printer_it->second(ptr);
235 } else {
236 ss << "[" << ptr.get() << "] : " << type_it->second << "]";
237 }
238 } else {
239 ss << "[" << ptr.get() << "] : unknown]";
240 }
241 } else {
242 ss << "[null]";
243 }
244 ss << std::endl;
245 }
246 return ss.str();
247 }

267 static void register_basic_printers()
268 {
269 register_printer<int>([](const int & v) {return std::to_string(v);});
270 register_printer<float>([](const float & v) {return std::to_string(v);});
271 register_printer<double>([](const double & v) {return std::to_string(v);});
272 register_printer<std::string>([](const std::string & v) {return v;});
273 register_printer<bool>([](const bool & v) {return v ? "true" : "false";});
