46inline std::string demangle(const char * name)
47{
48 int status = 0;
49 char * p = abi::__cxa_demangle(name, nullptr, nullptr, &status);
50 std::string out = (status == 0 && p) ? p : name;
51 std::free(p);
52 return out;
53}

59inline std::string stacktrace(std::size_t skip = 1, std::size_t max_frames = 64)
60{
61 void * buf[128];
62 const int n = backtrace(buf, static_cast<int>(std::min<std::size_t>(max_frames, 128)));
63 char ** syms = backtrace_symbols(buf, n);
64 std::ostringstream oss;
65 for (int i = static_cast<int>(skip); i < n; ++i) {
66 oss << "#" << (i - static_cast<int>(skip)) << " " << syms[i] << "\n";
67 }
68 std::free(syms);
69 return oss.str();
70}

103 template<typename T>
104 void set(const std::string & key, const T & value)
105 {
106 std::lock_guard<std::mutex> lock(state_mutex_);
107 auto it = values_.find(key);
108
109 if (it == values_.end()) {
110 values_[key] = std::make_shared<T>(value);
111 types_[key] = typeid(T).hash_code();
112 type_names_[key] = demangle(typeid(T).name());
113 } else {
114 if (types_[key] != typeid(T).hash_code()) {
115 std::ostringstream oss;
116 oss << "Type mismatch in set(\"" << key << "\")\n"
117 << " stored type : " << type_names_[key] << "\n"
118 << " provided type: " << demangle(typeid(T).name()) << "\n"
119 << "Backtrace:\n" << stacktrace(1);
120 throw std::runtime_error(oss.str());
121 }
122
123 auto ptr = std::static_pointer_cast<T>(it->second);
124 *ptr = value;
125 }
126 }

285 template<typename T>
286 const std::vector<std::shared_ptr<T>> get_group(const std::string & group_key) const
287 {
288 std::lock_guard<std::mutex> lock(group_mutex_);
289 auto group_it = groups_.find(group_key);
290
291 if (group_it == groups_.end()) {
292 throw std::runtime_error("Group key not found in get_group: " + group_key);
293 }
294
295 const auto group_keys = group_it->second;
296
297 std::vector<std::shared_ptr<T>> out;
298 out.reserve(group_keys.size());
299
300 for (const auto & group_key : group_keys) {
301 try {
302 out.push_back(get_ptr<T>(group_key));
303 } catch (const std::runtime_error & e) {
304 std::cerr << "Error retrieving key '" << group_key << "' from group '" << group_key <<
305 "': " << e.what() << std::endl;
306 }
307 }
308
309 return out;
310 }

369 template<typename T>
370 std::vector<std::shared_ptr<T>> get_no_group() const
371 {
372 // Snapshot the set of all keys that belong to any group (under group lock).
373 std::unordered_set<std::string> grouped_keys;
374 {
375 std::lock_guard<std::mutex> glock(group_mutex_);
376 for (const auto & g : groups_) {
377 for (const auto & k : g.second) {
378 grouped_keys.insert(k);
379 }
380 }
381 }
382
383 // Iterate values and return those of type T whose key is not grouped.
384 std::lock_guard<std::mutex> slock(state_mutex_);
385 std::vector<std::shared_ptr<T>> out;
386 const size_t target_hash = typeid(T).hash_code();
387 for (const auto & kv : values_) {
388 if (types_.at(kv.first) == target_hash &&
389 grouped_keys.find(kv.first) == grouped_keys.end())
390 {
391 out.push_back(std::static_pointer_cast<T>(kv.second));
392 }
393 }
394 return out;
395 }

423 template<typename T>
424 static void register_printer(std::function<std::string(const T &)> printer)
425 {
426 auto wrapper = [printer](std::shared_ptr<void> base_ptr) -> std::string {
427 auto typed_ptr = std::static_pointer_cast<T>(base_ptr);
428 return printer(*typed_ptr);
429 };
430 type_printers_[typeid(T).hash_code()] = wrapper;

438 std::string debug_string() const
439 {
440 std::stringstream ss;
441 for (const auto & kv : values_) {
442 ss << kv.first << " = ";
443 auto ptr = kv.second;
444 if (ptr) {
445 auto type_it = types_.find(kv.first);
446 if (type_it != types_.end()) {
447 auto printer_it = type_printers_.find(type_it->second);
448 if (printer_it != type_printers_.end()) {
449 ss << "[" << ptr.get() << "] : " << printer_it->second(ptr);
450 } else {
451 ss << "[" << ptr.get() << "] : " << type_it->second << "]";
452 }
453 } else {
454 ss << "[" << ptr.get() << "] : unknown]";
455 }
456 } else {
457 ss << "[null]";
458 }
459 ss << std::endl;
460 }
461 return ss.str();
462 }

482 static void register_basic_printers()
483 {
484 register_printer<int>([](const int & v) {return std::to_string(v);});
485 register_printer<float>([](const float & v) {return std::to_string(v);});
486 register_printer<double>([](const double & v) {return std::to_string(v);});
487 register_printer<std::string>([](const std::string & v) {return v;});
488 register_printer<bool>([](const bool & v) {return v ? "true" : "false";});
489 register_printer<char>([](const char & v) {return std::string(1, v);});
490 register_printer<std::vector<std::string>>(
491 [](const std::vector<std::string> & v) {
492 std::ostringstream oss;
493 oss << " " << v.size() << " [";
494 for (std::size_t i = 0; i < v.size(); ++i) {
495 if (i > 0) {oss << ", ";}
496 oss << v[i];

498 oss << "]";
