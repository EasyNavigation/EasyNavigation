63inline double footprint_radius(const std::vector<FootprintPoint> & footprint)
64{
65 double radius = 0.0;
66 for (const auto & p : footprint) {
67 radius = std::max(radius, std::hypot(p.x, p.y));
68 }
69 return radius;
70}

73inline bool footprint_contains(const std::vector<FootprintPoint> & footprint, double x, double y)
74{
75 bool inside = false;
76 for (size_t i = 0, j = footprint.size() - 1; i < footprint.size(); j = i++) {
77 const auto & a = footprint[i];
78 const auto & b = footprint[j];
79 if ((a.y > y) != (b.y > y) && x < (b.x - a.x) * (y - a.y) / (b.y - a.y) + a.x) {
80 inside = !inside;
81 }
82 }
83 return inside;
84}

87inline double footprint_border_distance(
88 const std::vector<FootprintPoint> & footprint, double x, double y)
89{
90 double best = std::numeric_limits<double>::infinity();
91 for (size_t i = 0, j = footprint.size() - 1; i < footprint.size(); j = i++) {
92 const auto & a = footprint[j];
93 const auto & b = footprint[i];
94 const double dx = b.x - a.x;
95 const double dy = b.y - a.y;
96 const double len2 = dx * dx + dy * dy;
97 const double t =
98 len2 > 0.0 ? std::clamp(((x - a.x) * dx + (y - a.y) * dy) / len2, 0.0, 1.0) : 0.0;
99 best = std::min(best, std::hypot(x - (a.x + t * dx), y - (a.y + t * dy)));
100 }
101 return best;
102}

106inline double footprint_inscribed_radius(const std::vector<FootprintPoint> & footprint)
107{
108 if (footprint.size() < 3 || !footprint_contains(footprint, 0.0, 0.0)) {
109 return 0.0;
110 }
111 return footprint_border_distance(footprint, 0.0, 0.0);
112}
