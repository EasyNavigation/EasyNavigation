142 bool pop(T & out)
143 {
144 std::lock_guard<std::mutex> lock(mutex_);
145 if (size_ == 0) {
146 return false;
147 }
148
149 out = buffer_[tail_];
150 tail_ = (tail_ + 1) % capacity_;
151 --size_;
152 return true;
153 }

159 bool latest(T & out) const
160 {
161 std::lock_guard<std::mutex> lock(mutex_);
162 if (size_ == 0) {
163 return false;
164 }
165
166 const std::size_t newest_index =
167 (head_ + capacity_ - 1) % capacity_;
168 out = buffer_[newest_index];
169 return true;
170 }

180 {
181 if (!buffer_[idx].engaged) {

190private:
192 void advance_head_()
193 {
194 head_ = (head_ + 1) % capacity_;
195 if (size_ < capacity_) {
196 ++size_;
197 } else {
198 // Buffer is full, we overwrite the oldest element.
199 tail_ = (tail_ + 1) % capacity_;
200 }
201 }
